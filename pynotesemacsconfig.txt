tab_pressed () {
    if (get_selection()) {
        indent_selection();
    }
    else {
        insert_text(tk_index("insert"), "\t");
    };
};

transpose_chars () {
    chars = get_text("insert-1c", "insert");
    delete_text("insert-1c", "insert");
    swapped = chars[::-1];
    insert_text("insert-1c", swapped);
};

kill_line () {
    line_number = tk_index("insert", "line");
    text_after_cursor = get_text(tk_index("insert"), line_number + ".end");
    if (text_after_cursor) {
        copy_clipboard(text_after_cursor);
        delete_text(tk_index("insert"), line_number + ".end");
    }
    else {
        delete_text(tk_index("insert"), line_number + ".end+1c");
    };
};

kill_whole_line () {
    line_number = tk_index("insert", "line");
    copy_clipboard(get_text(line_number + ".0", line_number + ".end+1c"));
    delete_text(line_number + ".0", line_number + ".end+1c");
};

unselect () {
    if (get_selection()) {
        select_text("0.0", "0.0");
    };
};

keyboard_quit () {
    remove_selection_point();
    unselect();
};

go_to_definition () {
    python_go_to_definition(ask_dialog("Variable name:"));
};

move_to_line_start () {
    move_cursor(tk_index("insert", "line") + ".0");
};

move_to_line_end () {
    move_cursor(tk_index("insert", "line") + ".end");
};

next_line () {
    move_cursor("insert+1l");
};

previous_line () {
    move_cursor("insert-1l");
};

forward_char () {
    move_cursor("insert+1c");
};

backward_char () {
    move_cursor("insert-1c");
};

(tabpressed) →:
(
if (getselection) {indentselection} else {insert tkindex ('insert'), '	'}
);

(transposechars) →:
(
setvar 'chars', get ('insert-1c', 'insert') ↩
delete 'insert-1c', 'insert' ↩
setvar 'swapped', chars[::-1] ↩
insert 'insert-1c', swapped
);

(killline) →:
(
setvar 'lineno', tkindex ('insert', 'line') ↩
setvar 'textaftercursor', get (tkindex ('insert'), lineno + '.end') ↩
if (textaftercursor) {
copytext textaftercursor ↓
delete tkindex ('insert'), lineno + '.end'
} else {
delete tkindex ('insert'), lineno + '.end+1c'
}
);

(killwholeline) →:
(
setvar 'lineno', tkindex ('insert', 'line') ↩
copytext get (lineno + '.0', lineno + '.end+1c') ↩
delete lineno + '.0', lineno + '.end+1c'
);

(unselect) →:
(
if (getselection) {select 0.0, 0.0}
);

(govardef) →:
(
pythongodef ask 'Variable name:'
);
